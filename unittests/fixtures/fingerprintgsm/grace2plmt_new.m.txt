%% GRACE2PLMT_NEW - Supplies synthetic GRACE data and observes recomputation options.
%
% Created by
%   2026/10/26, En-Chi Lee (williameclee@gmail.com)

function [p, s, dates] = grace2plmt_new(varargin)
    global DEGREE1_CONVERGENCE_FIXTURE
    v = DEGREE1_CONVERGENCE_FIXTURE;
    if ~isempty(varargin)
        ip = inputParser;
        ip.KeepUnmatched = true;
        addParameter(ip, 'RecomputeDegree1', false);
        parse(ip, varargin{4:end});
        DEGREE1_CONVERGENCE_FIXTURE.degree1Args = ip.Results.RecomputeDegree1;
        if ~isfield(v, 'readCount')
            v.readCount = 0;
        end
        DEGREE1_CONVERGENCE_FIXTURE.readCount = v.readCount + 1;
    end

    [order, degree] = addmon(2);
    ko = kernelorder(2);
    a = zeros(12, 3); a(ko, :) = v.x; a = reshape(a, 6, 2, 3);
    b = zeros(12, 3); b(ko, :) = v.sigma; b = reshape(b, 6, 2, 3);
    metadata = repmat([degree, order], 1, 1, 3);
    p = permute(cat(2, metadata, a), [3, 1, 2]);
    s = permute(cat(2, metadata, b), [3, 1, 2]);
    dates = datetime(2003, (1:3)', 15);
end
